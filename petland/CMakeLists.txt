# Copied by tools/build_petland.py into an isolated board-specific project.
cmake_minimum_required(VERSION 3.16)
file(TO_CMAKE_PATH "$ENV{WATCH_OS_COMMON_DIR}" WATCH_OS_COMMON)
file(TO_CMAKE_PATH "$ENV{WATCH_BOARD_PROJECT}" WATCH_BOARD_PROJECT)
if(NOT EXISTS "${WATCH_OS_COMMON}/petland/main/world.c" OR NOT EXISTS "$ENV{WATCH_BOARD_PROJECT}/components")
    message(FATAL_ERROR "Use tools/build_petland.py --board <firmware directory>")
endif()
set(EXTRA_COMPONENT_DIRS "${WATCH_BOARD_PROJECT}/components" "${WATCH_OS_COMMON}/components")
set(COMPONENTS main)
set(EXCLUDE_COMPONENTS "apps" "appkit" "services" "companion" "webradio" "hid_link" "watcher_camera" "bt_speaker")
set(PROJECT_VER "1.0.0")
include($ENV{IDF_PATH}/tools/cmake/project.cmake)
project(petland)
if(DEFINED ENV{PETLAND_DEV_CONSOLE})
    idf_component_get_property(petland_core core COMPONENT_LIB)
    target_compile_definitions(${petland_core} PRIVATE CONFIG_WATCH_DEV_CONSOLE=1)
endif()
